Pass by Reference in Rust 

Type                     | Write? | Read? | How Many?
-------------------------|--------|-------|-----------------------------
Immutable reference (&T) |   ❌    |  ✔    | Many (unlimited)
Mutable reference (&mut) |   ✔    |  ✔    | Only one at a time
Ownership (value)        |   ✔    |  ✔    | Only one owner
Slice (&[T], &str)       |   ❌    |  ✔    | Many (immutable borrow)
Mutable slice (&mut [T]) |   ✔    |  ✔    | Only one (mutable borrow)
Copy types               |   ✔    |  ✔    | Copied, not moved

 ✍️ // call by value example
fn main() {
let num1: i32=10;
let num2=20 ;: i32
println!("num1={}, num2={}",num1,num2);
swapbyvalue(num1, num2);
println!("num1={}, num2={}",num1,num2);

fn swapbyvalue(mut num1: i32,mut num2: i32){
num1=num1+num2;
num2=num1-num2;
num1=num1-num2;
println! ("num1={}, num2={}",num1,num2);
}

-----------------------------------------
✍️ // same example with call by referance
fn main() {
let mut num1: i32=10;
let mut num2: i32=20;
println! ("num1={}, num2={}",num1,num2);
swapbyref(&mut num1, &mut num2);
println!("num1={}, num2={}",num1,num2);

}

fn swapbyref(num1: &mut i32,num2: &mut i32){
*num1 =* num1+*num2;
*num2 =* num1 -* num2;
*num1 =* num1 -* num2;
println! ("num1={}, num2={}",num1,num2);

}

struct Circle {
    radius: f64,
}

trait Shape {
    fn area(&self) -> f64;
}

impl Shape for Circle {
    fn area(&self) -> f64 {
        3.14 * self.radius * self.radius
    }
}

fn main() {
    let circle: Circle = Circle { radius: 5.0 };
    let area = circle.area();
    println!("Circle area: {}", area);
}
