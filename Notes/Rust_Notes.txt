 ✍️ // call by value example
 
fn main() {
    let num1: i32 = 10;
    let num2: i32 = 20;
    println!("Before (main): num1 = {}, num2 = {}", num1, num2);

    swap_by_value(num1, num2); // passes copies

    println!("After  (main): num1 = {}, num2 = {}", num1, num2);
}

fn swap_by_value(mut num1: i32, mut num2: i32) {
    num1 = num1 + num2;
    num2 = num1 - num2;
    num1 = num1 - num2;

    println!("Inside function (by value): num1 = {}, num2 = {}", num1, num2);
}

fn main() {
    let reference_to_nothing = create_string_ref();
}

fn create_string_ref() -> &String {
    let s: String = String::from("hello");
    return &s;  // ← returning a reference to a local variable
}

--------------------
s is created inside create_string_ref.
When the function returns, s is dropped (its memory is freed).
Returning &s would point to freed memory → dangling reference.
Rust’s borrow checker rejects this at compile time with an error like:

returns a reference to data owned by the current function

struct Circle {
    radius: f64,
}

trait Shape {
    fn area(&self) -> f64;
}

impl Shape for Circle {
    fn area(&self) -> f64 {
        3.14 * self.radius * self.radius
    }
}
